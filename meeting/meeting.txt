\section{Meeting: 25 November 2026}

\begin{frame}{Nowcasting Model: Offset}

Model for the intensity of $Y_t$, the true number of events that occured at $t$

\[Y_t \sim \operatorname{NB}(\lambda_t, \phi)\]

\[log(\lambda_t) = \beta_o + s(t) \]
    
\end{frame}


\begin{frame}{Nowcasting Model: Offset}

$C_{t,d}$ : Cumulated number of events that occurred at $t$ but reported by $t+d$. \\
$Y_t$ : True number of events that occurred at $t$. \\
$F_t(d)$ : Reporting delay distribution function. 

\begin{equation}
     E(C_{t,d}) = F_{t}(d)E(Y_{t})
\end{equation}

If we let $\mu_{t,d} = E(C_{t,d})$ then

\[ log(\mu_{t,d}) = log(\lambda_t) + O_{t,d}\] where $O_{t,d} = log(F_{t}(d))$
    
\end{frame}

\begin{frame}{Nowcasting Model: Offset}

    \begin{itemize}
        \item Problem: We can't use the intensity model to predict the $Y_t$ because of the constraint that $C_{t,T-t} \le \hat{\lambda}_t$.
        \item Solution: Since we have observed $C_{t,T-t}$ we write (1) as \[C_{t,T-t} = F_{t}(T-t)E(Y_{t})\]
    \end{itemize}
\end{frame}
